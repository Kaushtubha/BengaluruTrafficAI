\chapter{Backend Architecture and API Lifecycle}

\section{Flask Web Framework and Execution Model}
The backend is built on **Flask 3.1**, implementing lightweight REST API routing, request validation, and JSON response serialization. Cross-Origin Resource Sharing (CORS) is enabled via \texttt{flask-cors} to permit secure communication from the Vercel-hosted frontend domain.

\section{Endpoint Specifications and Request Lifecycles}

\subsection{\texttt{GET /api/health}}
\begin{itemize}[leftmargin=*]
    \item \textbf{Purpose}: Service liveness probe and uptime monitor utilized by hosting providers and health checks.
    \item \textbf{Response Payload}:
\begin{lstlisting}[language=json]
{
  "status": "ok",
  "mode": "demo", // or "live_cv"
  "uptime_seconds": 3600
}
\end{lstlisting}
\end{itemize}

\subsection{\texttt{GET /api/driver\_advice}}
\begin{itemize}[leftmargin=*]
    \item \textbf{Purpose}: Commuter guidance endpoint providing corridor queue counts, signal phase recommendations, and straight-line Haversine distance telemetry.
    \item \textbf{Query Parameter Validation}:
    \begin{itemize}
        \item \texttt{corridor} (string, optional): Evaluated case-insensitively against \texttt{["North", "South", "East", "West"]}. Strings exceeding 32 characters or invalid lane names trigger an immediate \texttt{HTTP 400 Bad Request}.
        \item \texttt{lat}, \texttt{lon} (float, optional): Validated to ensure numeric format and valid geographic boundaries ($-90.0 \le \text{lat} \le 90.0$ and $-180.0 \le \text{lon} \le 180.0$). If only one coordinate is provided, the API sets the distance status to \texttt{INCOMPLETE\_COORDINATES\_MISSING\_LAT} or \texttt{INCOMPLETE\_COORDINATES\_MISSING\_LON}.
    \end{itemize}
    \item \textbf{Response Schema}:
\begin{lstlisting}[language=json]
{
  "status": "success",
  "data_source": "demo_sim", // or "live_cv"
  "feed_description": "Render Free-Tier Simulation (Demonstration Data)",
  "junction": {
    "name": "Silk Board, Bengaluru",
    "latitude": 12.9176,
    "longitude": 77.6238,
    "is_configurable": true,
    "configuration_error": null
  },
  "approach": {
    "corridor": "North",
    "visible_vehicle_count": 12,
    "total_junction_vehicles": 43,
    "assigned_green_time_sec": 28.3,
    "cycle_time_sec": 100.0,
    "congestion_level": "Medium",
    "congestion_provenance": "synthetic_demo_rule"
  },
  "distance_telemetry": {
    "calculation_method": "Haversine Geodesic (Straight-Line)",
    "straight_line_distance_km": 1.96,
    "status": "CALCULATED",
    "disclaimer": "Straight-line aerial distance only. Actual road driving distance will be longer due to road topology and flyover ramps."
  },
  "delay_telemetry": {
    "delay_status": "ESTIMATE_UNAVAILABLE",
    "estimated_delay_mins": null,
    "reason": "Queue clearance delay cannot be accurately calculated without upstream queue length and continuous arrival rates outside camera FOV."
  },
  "safety_guidelines": [
    "Approaching North corridor. Maintain safe braking distance in stop-and-go queue.",

  ],
  "corridor_status": {
    "monitored_corridors": ["North", "South", "East", "West"],
    "unmonitored_routes": "DATA_UNAVAILABLE"
  }
}
\end{lstlisting}
\end{itemize}

\subsection{\texttt{POST /api/simulate\_emergency}}
\begin{itemize}[leftmargin=*]
    \item \textbf{Purpose}: Operator actuation endpoint allowing simulation and clearing of emergency vehicle priority overrides.
    \item \textbf{Request Body}: \texttt{\{"lane": "North"\}} triggers override; \texttt{\{"lane": null\}} clears active override.
\end{itemize}
