\section{Representative Regression Test Excerpts}

\begin{lstlisting}[language=Python,caption={Proportional Signal Timing Bounds Test Excerpt}]
def test_proportional_allocation_bounds(self):
    counts = {"North": 30, "South": 10, "East": 5, "West": 5}
    plan = get_signal_plan(counts)
    # Busiest lane must receive the priority allocation
    self.assertEqual(plan["priority_lane"], "North")
    for lane, seconds in plan["green_times"].items():
        self.assertGreaterEqual(seconds, MIN_GREEN)
        self.assertLessEqual(seconds, MAX_GREEN)
    self.assertEqual(plan["cycle_time"], round(sum(plan["green_times"].values()), 1))
\end{lstlisting}

\begin{lstlisting}[language=Python,caption={Speed Estimation Gap Reset Test Excerpt}]
def test_stale_tracking_gap_resets(self):
    # If a track disappears for > 3 seconds, calculation must reset
    update_speed(track_id=101, cx=100, cy=100, timestamp=10.0)
    update_speed(track_id=101, cx=150, cy=150, timestamp=10.1)
    # Gap of 4.0 seconds (> MAX_TRACK_AGE_SECONDS = 3.0)
    update_speed(track_id=101, cx=200, cy=200, timestamp=14.1)
    speed = calculate_speed(track_id=101)
    # Only 1 observation remains in the new window; returns safe 0.0
    self.assertEqual(speed, 0.0)
\end{lstlisting}
