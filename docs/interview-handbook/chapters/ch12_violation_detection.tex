\chapter{Traffic-Violation Detection}

\section{Violation Detection Architecture}
Traffic infraction monitoring is divided across four specialized detection modules within the \texttt{ai/} package:
\begin{enumerate}
    \item \textbf{Red Light Jump Detection} (\texttt{ai/violation\_detector.py})
    \item \textbf{Speed Limit Violation} (\texttt{ai/speed\_estimator.py})
    \item \textbf{Wrong-Side Driving Analysis} (\texttt{ai/wrong\_side\_detector.py})
    \item \textbf{Motorcycle Helmet Compliance} (\texttt{ai/helmet\_detector.py})
\end{enumerate}

\section{Module Implementations and Deduplication Strategies}

\subsection{1. Stop-Line Red Light Violations}
\begin{itemize}[leftmargin=*]
    \item \textbf{Condition}: Triggered when the current signal state is \texttt{"RED"} and the vehicle's bottom bounding coordinate exceeds the predefined horizontal virtual stop-line:
    \begin{equation}
    \text{Violation} = (\text{signal} == \text{"RED"}) \land (y_2 > y_{\text{stop\_line}}) \quad \text{where } y_{\text{stop\_line}} = 300\,\text{px}
    \end{equation}
    \item \textbf{Deduplication}: To avoid repeatedly logging the same vehicle as it traverses past the line, the detector maintains a sliding history of the last 20 logged track IDs:
\begin{lstlisting}[language=Python]
recent_ids = [v["track_id"] for v in violations[-20:] if v.get("track_id") is not None]
if track_id is not None and track_id in recent_ids:
    return False, None
\end{lstlisting}
\end{itemize}

\subsection{2. Directional Wrong-Side Counterflow}
\begin{itemize}[leftmargin=*]
    \item \textbf{Geometry and Expectation}: In typical dual-carriageway traffic, vehicles in the left half of the video frame ($x < W/2$) are expected to travel southwards ($cy$ increasing). Vehicles in the right half ($x \ge W/2$) are expected to travel northwards ($cy$ decreasing).
    \item \textbf{Vector Calculation}: Evaluates historical centroid movement across 5 position snapshots:
    \begin{equation}
    \Delta y = y_{\text{latest}} - y_{\text{earliest}}
    \end{equation}
    If in the left lane and $\Delta y < -20\,\text{px}$ (moving upwards), or in the right lane and $\Delta y > 20\,\text{px}$ (moving downwards), a counterflow flag is raised.
    \item \textbf{Deduplication}: Track IDs are recorded in \texttt{wrong\_side\_violations}; once flagged, a vehicle is never re-flagged.
\end{itemize}

\subsection{3. Motorcycle Helmet Compliance}
\begin{itemize}[leftmargin=*]
    \item \textbf{Implementation}: Simulated compliance heuristic operating on vehicles classified by YOLOv8 as \texttt{"Motorcycle"}.
    \item \textbf{Deterministic Rule}: Evaluates riders where $\text{track\_id} \pmod 7 == 0$ ($\approx 14.3\%$ violation rate) to maintain stable simulation without random noise jittering. Deduplicated using the recent 10 logged track IDs.
\end{itemize}
