\chapter{Vehicle Tracking}

\section{The Multi-Object Tracking (MOT) Problem}
Frame-by-frame object detection produces independent collections of bounding boxes. Object detection alone has no concept of temporal identity: bounding box \#1 in frame $t$ cannot be linked to bounding box \#4 in frame $t+1$.

Without persistent identity:
\begin{enumerate}
    \item A single car idling at a red light for 30 seconds would register as 900 distinct vehicles ($30\,\text{s} \times 30\,\text{FPS}$), distorting total volume telemetry.
    \item Velocity calculation ($v = \Delta d / \Delta t$) is impossible because the displacement of an individual object over time cannot be tracked.
    \item Traffic violations would trigger duplicates on every consecutive frame.
\end{enumerate}

\section{Pure-Python IoU Tracker Algorithm}
To avoid heavy external C++ compiled dependencies (such as C-compiled Kalman filters or deep cosine embedding extractors), the repository implements a lightweight greedy IoU tracker in \texttt{backend/tracker\_manager.py}.

\subsection{Intersection over Union (IoU) Formulation}
Given track bounding box $A = [x_1^A, y_1^A, x_2^A, y_2^A]$ and detection bounding box $B = [x_1^B, y_1^B, x_2^B, y_2^B]$:
\begin{align}
x_{\text{left}} &= \max(x_1^A, x_1^B), \quad y_{\text{top}} = \max(y_1^A, y_1^B) \\
x_{\text{right}} &= \min(x_2^A, x_2^B), \quad y_{\text{bottom}} = \min(y_2^A, y_2^B) \\
\text{Area}_{\text{inter}} &= \max(0, x_{\text{right}} - x_{\text{left}}) \times \max(0, y_{\text{bottom}} - y_{\text{top}}) \\
\text{Area}_{\text{union}} &= \text{Area}_A + \text{Area}_B - \text{Area}_{\text{inter}} \\
\text{IoU}(A, B) &= \frac{\text{Area}_{\text{inter}}}{\text{Area}_{\text{union}}} \in [0.0, 1.0]
\end{align}

\subsection{Greedy Association Logic}
\begin{lstlisting}[language=Python,caption={IoU Matching Engine (\texttt{backend/tracker\_manager.py})}]
for track in self.tracks:
    track.age += 1
    best_iou = 0
    best_det_idx = -1

    for idx, det_bbox in enumerate(det_bboxes):
        if idx in matched_det_indices:
            continue
        iou = self.calculate_iou(track.bbox, det_bbox)
        if iou > best_iou:
            best_iou = iou
            best_det_idx = idx

    if best_iou > self.iou_threshold:  # iou_threshold = 0.2
        track.bbox = det_bboxes[best_det_idx]
        track.hits += 1
        track.age = 0  # reset lost-frame counter
        matched_det_indices.add(best_det_idx)
        updated_tracks.append(track)
    else:
        if track.age <= self.max_lost:  # max_lost = 10 frames
            updated_tracks.append(track)

# Spawn new tracks for unmatched detections
for idx, det_bbox in enumerate(det_bboxes):
    if idx not in matched_det_indices:
        new_track = Track(self.next_id, det_bbox, det_labels[idx])
        self.next_id += 1
        updated_tracks.append(new_track)
\end{lstlisting}

\section{Algorithmic Trade-offs: IoU vs. DeepSORT and ByteTrack}
\begin{itemize}[leftmargin=*]
    \item \textbf{Computational Footprint}: The pure-Python IoU tracker executes in $< 0.1\,\text{ms}$ per frame with $O(N \cdot M)$ complexity. DeepSORT requires generating a 128-dimensional appearance embedding per detection via a secondary CNN, increasing latency tenfold.
    \item \textbf{Failure Mode under Occlusion}: When a vehicle is completely occluded by a large bus for $> 10$ frames ($\approx 0.33\,\text{s}$), the IoU tracker drops the track and assigns a new ID upon reappearance. DeepSORT or ByteTrack retain tracks longer using linear velocity Kalman extrapolation and visual feature re-identification.
\end{itemize}
